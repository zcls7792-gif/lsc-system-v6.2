步骤 4: 验证覆盖率指标
# 快速检查行覆盖率
python3 -c "
import xml.etree.ElementTree as ET
tree = ET.parse('target/jacoco/jacoco.xml')
root = tree.getroot()
for c in root.iter('counter'):
    if c.get('type') == 'LINE':
        missed = int(c.get('missed'))
        covered = int(c.get('covered'))
        rate = covered / (missed + covered) * 100
        print(f'LINE: {covered}/{missed+covered} = {rate:.1f}%')
"

# 检查安全包覆盖率
python3 -c "
import xml.etree.ElementTree as ET
tree = ET.parse('target/jacoco/jacoco.xml')
root = tree.getroot()
for pkg in root.iter('package'):
    if 'security' in pkg.get('name', ''):
        for c in pkg.iter('counter'):
            if c.get('type') == 'LINE':
                m = int(c.get('missed'))
                cov = int(c.get('covered'))
                print(f'security: LINE {cov}/{m+cov} = {cov/(m+cov)*100:.1f}%')
"
